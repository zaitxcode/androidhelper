📦 Installation & Setup
apputilx setup & Utils.initialize(this)
Add the library dependency to your build file then initialize inside Application class.
// 1. Add in gradle/libs.versions.toml
[versions]
apputilx = "1.5.0-alpha01"

[libraries]
apputilx = { group = "com.github.mohamed-zaitoon", name = "apputilx", version.ref = "apputilx" }

// 2. Add in settings.gradle.kts
repositories {
    google()
    mavenCentral()
    maven { url = uri("https://jitpack.io") }
}

// 3. Add in app/build.gradle.kts
dependencies {
    implementation(libs.apputilx)
}

// 4. Initialize in Application class
class ExampleApplication : Application() {
    override fun onCreate() {
        super.onCreate()
        Utils.initialize(this)
    }
}

🌐 Network Helpers
Utils.isConnected & networkState
Check internet connectivity, validated connection, and transport type.
// Kotlin
val isOnline: Boolean = Utils.isConnected
val isValidated: Boolean = Utils.hasValidatedInternet()
val isMetered: Boolean = Utils.isConnectionMetered()
// Flutter (Dart)
final isOnline = await _channel.invokeMethod<bool>('isConnected');
final stateDetails = await _channel.invokeMethod<String>('networkState');
🔗 Intents & Actions
Utils.openUrl, openWhatsApp, dial, sendSms, sendEmail
Open URLs, WhatsApp chats, dialer, SMS, and send emails directly.
// Kotlin
Utils.openUrl(context, "https://apputilx.mohamedzaitoon.com")
Utils.openWhatsApp("201234567890", "Hello")
Utils.dial("201234567890")
Utils.sendSms("201234567890", "Test message")
Utils.openPlayStore()
// Flutter (Dart)
await _channel.invokeMethod('openUrl', {'url': 'https://apputilx.mohamedzaitoon.com'});
await _channel.invokeMethod('openWhatsApp', {'text': 'Hello'});
await _channel.invokeMethod('dial');
await _channel.invokeMethod('openPlayStore');
📋 Clipboard, Vibration & Screen
Utils.vibrate & copyText & blockCapture
Trigger physical vibration motor (Android 13-16 ready), clipboard, and screen capture block.
// Kotlin
Utils.copyText("Copied text")
Utils.vibrate(200)
Utils.vibratePattern(longArrayOf(0, 100, 50, 200), -1)
Utils.blockCapture()
// Flutter (Dart)
await _channel.invokeMethod('copyText', {'text': 'Copied'});
await _channel.invokeMethod('vibrate', {'ms': 200});
await _channel.invokeMethod('blockCapture');
🔔 Notifications & Keyboard
Utils.showNotification & hideKeyboard
Create notification channels, show custom notifications, and hide input keyboard.
// Kotlin
Utils.createNotificationChannel("demo", "Demo Channel")
Utils.showNotification(
    channelId = "demo",
    title = "AppUtilX",
    text = "Hello Notification",
    iconResId = R.drawable.ic_launcher
)
Utils.hideKeyboard()
// Flutter (Dart)
await _channel.invokeMethod('showNotification', {'text': 'Flutter Notification'});
await _channel.invokeMethod('hideKeyboard');
📱 Device, Battery & App Info
Utils.deviceName, getBatteryLevel, appInfo
Query device model, battery status/charging type, and foreground application state.
// Kotlin
val name = Utils.deviceName()
val battery = Utils.getBatteryLevel()
val isCharging = Utils.isCharging()
val isForeground = Utils.isAppInForeground()
// Flutter (Dart)
final devInfo = await _channel.invokeMethod<String>('deviceInfo');
final batInfo = await _channel.invokeMethod<String>('batteryInfo');
final appState = await _channel.invokeMethod<String>('appState');
💾 Data, Storage & Crypto
Utils.writeFile, readFile, sha256, base64Encode
File write/read/delete operations, SHA-256, SHA-512, HMAC, and Base64 encoding.
// Kotlin
Utils.writeFile("demo.txt", "content")
val content = Utils.readFile("demo.txt")
val hash = Utils.sha256("password")
val base64 = Utils.base64Encode("Hello")
// Flutter (Dart)
await _channel.invokeMethod('writeFile', {'text': 'content'});
final content = await _channel.invokeMethod<String>('readFile');
final hash = await _channel.invokeMethod<String>('sha256', {'text': 'password'});

        title = "Authenticate",
        subtitle = "Confirm identity",
        onSuccess = { /* Success */ }
    )
}
val sha1 = Utils.getPrimarySignatureSHA1()
// Flutter (Dart)
final result = await _channel.invokeMethod<String>('biometric');
final sig = await _channel.invokeMethod<String>('signatures');
